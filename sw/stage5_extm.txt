# 段 5 の EXTM_H* の手順 (= TB の +extm_cmd。書き方は tb/tb_mem.v の「EXTM_H* のマスタのモデル」)
# 「# 」で始まる行は読まない。アドレスとデータは 16 進、HSIZE は 10 進 (= 0 / 1 / 2 / 3)
#
# 1. プログラムが置いた値を読む (= プログラムが「EXTM 1」の行を出してから)
wait
# DDR の 0x0000_9000〜 (= プログラムが 0xA5A5_0000 + i を 8 word 書いた)
r 00009000 3 A5A50001A5A50000
r 0000900C 2 A5A5000300000000
r 00009010 2 00000000A5A50004
r 0000901C 0 0000000700000000
# DLM の 0xA020_1000〜 (= プログラムが 0xD1D1_D000 + i を 4 word 書いた)
r A0201008 3 D1D1D003D1D1D002
r A0201002 1 00000000D1D10000
#
# 2. プログラムが読む値を書く
# DDR: 32 bit を下側と上側に 1 つずつ、64 bit を 1 つ
w 00009100 2 0000000011223344
w 00009104 2 5566778800000000
w 00009108 3 0123456789ABCDEF
# DLM: 8 bit を 2 つ (= レーン 0 と 7)、16 bit を 1 つ (= レーン 2〜3)
w A0201100 0 00000000000000AB
w A0201107 0 CD00000000000000
w A0201102 1 00000000BEEF0000
# EXTS_H* の RAM: 64 bit の上側 (= アドレスの bit 2 が 1) と下側に 32 bit を 1 つずつ
w E8000104 2 CAFEBABE00000000
w E8000100 2 00000000FACEFEED
# ILM
w A000F000 2 0000000012345678
# PLIC の priority[20] (= EXTM_H* から AE350 の中のレジスタにも書ける)
w E4000050 2 0000000000000002
#
# 3. エラー応答
# 割り当ての無いアドレス、ROM への書き込み、APB のどこにも当たらないアドレス、32 bit の行き先への 64 bit の転送
re C0000000 2
we 80000000 2 0000000000000000
we F8002000 2 0000000000000000
re E8000000 3
re F8001000 3
re E4000004 3
re E6000000 3
re 80000000 3
#
# 4. APB の先 (= GP_INT のレジスタ。プログラムは触っていないので 0)
r F8001000 2 0000000000000000
#
# 5. 最後に合図を書く (= プログラムはこれを待ってから、2 の値を確かめる)
w A0201F00 2 000000000000600D
